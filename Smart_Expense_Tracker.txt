import tkinter as tk
from tkinter import ttk, messagebox
import csv
import os

FILE = "expenses.csv"

def add_expense():
    category = category_box.get()
    amount = amount_entry.get()
    note = note_entry.get()

    if not category or not amount:
        messagebox.showwarning("Warning", "Please fill all required fields")
        return

    try:
        amount = float(amount)
    except ValueError:
        messagebox.showerror("Error", "Enter a valid amount")
        return

    file_exists = os.path.exists(FILE)

    with open(FILE, "a", newline="") as file:
        writer = csv.writer(file)

        if not file_exists:
            writer.writerow(["Category", "Amount", "Note"])

        writer.writerow([category, amount, note])

    tree.insert("", "end", values=(category, amount, note))

    amount_entry.delete(0, tk.END)
    note_entry.delete(0, tk.END)

    update_total()


def update_total():
    total = 0

    for item in tree.get_children():
        total += float(tree.item(item)["values"][1])

    total_label.config(text=f"Total Expense: ₹{total:.2f}")


def load_data():
    if os.path.exists(FILE):
        with open(FILE, "r") as file:
            reader = csv.reader(file)
            next(reader, None)

            for row in reader:
                tree.insert("", "end", values=row)

        update_total()


root = tk.Tk()
root.title("Smart Expense Tracker")
root.geometry("700x500")

title = tk.Label(
    root,
    text="💰 Smart Expense Tracker",
    font=("Arial", 24, "bold")
)
title.pack(pady=20)

frame = tk.Frame(root)
frame.pack()

tk.Label(frame, text="Category").grid(row=0, column=0, padx=5)
category_box = ttk.Combobox(
    frame,
    values=["Food", "Travel", "Shopping", "Education", "Bills", "Other"]
)
category_box.grid(row=0, column=1)

tk.Label(frame, text="Amount").grid(row=1, column=0, padx=5, pady=10)
amount_entry = tk.Entry(frame)
amount_entry.grid(row=1, column=1)

tk.Label(frame, text="Note").grid(row=2, column=0)
note_entry = tk.Entry(frame)
note_entry.grid(row=2, column=1)

tk.Button(
    frame,
    text="Add Expense",
    command=add_expense
).grid(row=3, column=1, pady=15)

columns = ("Category", "Amount", "Note")

tree = ttk.Treeview(root, columns=columns, show="headings")

for col in columns:
    tree.heading(col, text=col)

tree.pack(fill="both", expand=True, padx=20)

total_label = tk.Label(
    root,
    text="Total Expense: ₹0",
    font=("Arial", 16, "bold")
)
total_label.pack(pady=10)

load_data()

root.mainloop()
